cmake_minimum_required(VERSION 3.16)
project(UltimateAutoClicker VERSION 1.2.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Use build.bat and test.bat on Windows")
endif()
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk+-3.0)
pkg_check_modules(X11 REQUIRED IMPORTED_TARGET x11 xtst)
find_package(Threads REQUIRED)
add_library(core src/core.cpp)
target_include_directories(core PUBLIC src)
target_link_libraries(core PUBLIC Threads::Threads)
target_compile_options(core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ultimate-autoclicker src/linux.cpp)
target_link_libraries(ultimate-autoclicker PRIVATE core PkgConfig::GTK PkgConfig::X11)
target_compile_options(ultimate-autoclicker PRIVATE -Wall -Wextra -Wpedantic)
enable_testing()
add_executable(core-tests tests/tests.cpp)
target_link_libraries(core-tests PRIVATE core)
add_test(NAME core COMMAND core-tests)
find_program(XVFB_RUN xvfb-run)
if(XVFB_RUN)
    add_test(NAME linux-integration COMMAND ${XVFB_RUN} -a $<TARGET_FILE:ultimate-autoclicker> --selftest)
endif()
install(TARGETS ultimate-autoclicker RUNTIME DESTINATION bin)
install(FILES assets/ultimate-autoclicker.desktop DESTINATION share/applications)
install(FILES LICENSE README.md DESTINATION share/doc/ultimate-autoclicker)
set(CPACK_PACKAGE_NAME ultimate-autoclicker)
set(CPACK_PACKAGE_CONTACT "https://github.com/kuy124/UltimateAutoClicker/issues")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native X11 mouse automation with a GTK interface")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_GENERATOR "TGZ;TXZ;ZIP;DEB;RPM")
set(CPACK_PACKAGE_FILE_NAME "UltimateAutoClicker-${PROJECT_VERSION}-Linux-${CMAKE_SYSTEM_PROCESSOR}")
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_RPM_PACKAGE_LICENSE MIT)
set(CPACK_RPM_EXCLUDE_FROM_AUTO_FILELIST_ADDITION /usr/share/applications)
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
include(CPack)
